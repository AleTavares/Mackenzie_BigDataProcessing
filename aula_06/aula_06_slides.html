🔍 Check de Completude — Nulls e Missing

Verificar se campos obrigatórios estão preenchidos. Campos opcionais podem ter null, mas campos de negócio críticos não.
from pyspark.sql.functions import col, count, when, isnan, lit

def check_completeness(df, required_columns, threshold=0.95):
    """
    Verifica completude dos campos obrigatórios.
    threshold=0.95 → aceita até 5% de nulls.
    """
    total = df.count()
    results = []
    
    for column in required_columns:
        null_count = df.filter(
            col(column).isNull() | (col(column) == "") | isnan(col(column))
        ).count()
        
        completeness = 1 - (null_count / total)
        passed = completeness >= threshold
        
        results.append({
            "column": column,
            "total_rows": total,
            "null_count": null_count,
            "completeness": round(completeness, 4),
            "threshold": threshold,
            "passed": passed
        })
    
    return results

# Uso
required = ["order_id", "customer_id", "total_amount", "order_date"]
results = check_completeness(df_bronze, required, threshold=0.98)

🔄 Check de Unicidade — Duplicatas

Verificar se registros com a mesma chave primária aparecem mais de uma vez. Duplicatas inflam métricas e causam double-counting.
from pyspark.sql.functions import count as spark_count, row_number
from pyspark.sql.window import Window

def check_uniqueness(df, key_columns):
    """Identifica duplicatas baseado nas colunas-chave."""
    total = df.count()
    
    # Contar registros por chave
    duplicates = df.groupBy(key_columns) \
        .agg(spark_count("*").alias("occurrences")) \
        .filter(col("occurrences") > 1)
    
    dup_keys = duplicates.count()
    dup_records = duplicates.agg(
        spark_count("*").alias("dup_keys"),
        {"occurrences": "sum"}
    ).collect()[0]
    
    return {
        "total_rows": total,
        "duplicate_keys": dup_keys,
        "uniqueness_ratio": round(1 - (dup_keys / total), 4),
        "passed": dup_keys == 0
    }

# Parceiro B: verificar unicidade por order_id
result = check_uniqueness(df_parceiro_b, key_columns=["order_id"])
# → {"duplicate_keys": 3200, "uniqueness_ratio": 0.968, "passed": False}

🔗 Integridade Referencial — Orphan Records

Verificar se valores em uma coluna existem como chave primária em outra tabela. Registros "órfãos" indicam inconsistência entre fontes.
def check_referential_integrity(df_child, df_parent, 
                                  child_col, parent_col):
    """
    Verifica se todos os valores de child_col existem em parent_col.
    Retorna registros órfãos (sem pai).
    """
    # Left anti join: registros em child que NÃO existem em parent
    df_orphans = df_child.join(
        df_parent.select(col(parent_col).alias("_ref")),
        df_child[child_col] == col("_ref"),
        "left_anti"
    )
    
    orphan_count = df_orphans.count()
    total = df_child.count()
    
    return {
        "total_rows": total,
        "orphan_count": orphan_count,
        "integrity_ratio": round(1 - (orphan_count / total), 4),
        "passed": orphan_count == 0,
        "orphan_df": df_orphans
    }

# Parceiro C: pedidos referenciando clientes inexistentes
result = check_referential_integrity(
    df_pedidos, df_clientes,
    child_col="customer_id", parent_col="id"

🔧 Quarentena — Implementação

from pyspark.sql.functions import array, when, lit, current_timestamp

def quarantine_invalid_records(df, rules, output_path):
    """Separa dados válidos/inválidos e salva quarentena."""
    
    # Adicionar coluna com lista de regras falhadas
    failed_rules_col = array(*[
        when(~condition, lit(rule_name))
        for rule_name, condition in rules.items()
    ])
    
    df_flagged = df.withColumn("_failed_rules", failed_rules_col)
    
    # Válidos: nenhuma regra falhou (array vazio)
    df_valid = df_flagged.filter(
        size(col("_failed_rules")) == 0
    ).drop("_failed_rules")
    
    # Inválidos: pelo menos 1 regra falhou
    df_quarantine = df_flagged.filter(
        size(col("_failed_rules")) > 0
    ).withColumn("_rejection_ts", current_timestamp()) \
     .withColumn("_source", lit("parceiro_a"))
    
    # Persistir quarentena
    df_quarantine.write.mode("append") \
        .partitionBy("order_date") \
        .parquet(f"{output_path}/quarantine/")
    
    return df_valid, df_quarantine.count()

df_clean, rejected = quarantine_invalid_records(
    df_bronze, validity_rules, "datalake"
)

def generate_quality_report(df, rules, key_columns, reference_dfs=None):
    """Gera relatório completo de qualidade."""
    total = df.count()
    report = {
        "execution_date": datetime.now().isoformat(),
        "total_records": total,
        "checks": {}
    }
    
    # 1. Completude
    for col_name in required_columns:
        nulls = df.filter(col(col_name).isNull()).count()
        report["checks"][f"completeness_{col_name}"] = {
            "metric": round(1 - nulls/total, 4),
            "threshold": 0.98,
            "passed": (1 - nulls/total) >= 0.98
        }
    
    # 2. Unicidade
    uniq = df.dropDuplicates(key_columns).count()
    report["checks"]["uniqueness"] = {
        "metric": round(uniq/total, 4),
        "threshold": 0.99,
        "passed": (uniq/total) >= 0.99
    }
    
    # 3. Validade (por regra)
    for rule_name, condition in rules.items():
        valid = df.filter(condition).count()
        report["checks"][f"validity_{rule_name}"] = {
            "metric": round(valid/total, 4),
            "threshold": 0.95,
            "passed": (valid/total) >= 0.95
        }
    
    # Score geral
    all_passed = all(c["passed"] for c in report["checks"].values())
    report["overall_passed"] = all_passed
    return report

from airflow import DAG
from airflow.operators.python import PythonOperator, BranchPythonOperator

def run_quality_checks(**kwargs):
    """Executa todos os checks e decide se continua."""
    report = generate_quality_report(df_bronze, rules, ["order_id"])
    
    # Push report para XCom
    kwargs["ti"].xcom_push(key="quality_report", value=report)
    
    return report["overall_passed"]

def decide_on_quality(**kwargs):
    """Branch: se qualidade OK → Silver, senão → alerta."""
    ti = kwargs["ti"]
    passed = ti.xcom_pull(task_ids="quality_checks")
    
    if passed:
        return "transform_silver"
    else:
        return "alert_quality_failure"

with DAG("pipeline_with_quality", ...) as dag:
    ingest = PythonOperator(task_id="ingest_bronze", ...)
    
    quality = PythonOperator(
        task_id="quality_checks",
        python_callable=run_quality_checks
    )
    
    branch = BranchPythonOperator(
        task_id="quality_gate",
        python_callable=decide_on_quality
    )
    
    transform = SparkSubmitOperator(task_id="transform_silver", ...)
    alert = PythonOperator(task_id="alert_quality_failure", ...)
    
    ingest >> quality >> branch >> [transform, alert]

🚦 Quality Gate Pattern — Bloqueio Inteligente

QUALITY GATE — 3 modos:

1. STRICT (Gold layer):
   Score < threshold → BLOQUEIA pipeline + ALERTA
   Nenhum dado ruim passa.

2. LENIENT (Silver layer):
   Score < threshold → QUARENTENA + CONTINUA
   Dados ruins são isolados, bons seguem.

3. MONITOR (Bronze layer):
   Score < threshold → LOG + MÉTRICA
   Apenas observar, não bloquear.
CamadaModoAção em Falha
Bronze → SilverLENIENTQuarentena + continua
Silver → GoldSTRICTBloqueia + alerta Roberto
Gold → DashboardSTRICTBloqueia publicação
Regra DataFlow: "Dados ruins NUNCA chegam na Gold. Se quality gate falha na Silver→Gold, pipeline para e Carlos é notificado. Melhor não publicar do que publicar errado."
