    parser.add_argument("--input", default="/datalake/bronze/",

    return parser.parse_args()

def main():
    args = parse_args()
    spark = SparkSession.builder.appName(f"silver_{args.date}").getOrCreate()
    
    # Lógica determinística, parametrizada
    transform_silver(spark, args.date, args.input, args.output, args.partner)

if __name__ == "__main__":
    main()

import logging
import json
from datetime import datetime

class JsonFormatter(logging.Formatter):
    """Formatter que emite logs como JSON."""
    def format(self, record):
        log_entry = {
            "timestamp": datetime.utcnow().isoformat() + "Z",
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno
        }
        # Adicionar campos extras (contexto)
        if hasattr(record, "extra_fields"):
            log_entry.update(record.extra_fields)
        return json.dumps(log_entry)

def get_logger(name, level=logging.INFO):
    """Retorna logger configurado com JSON output."""
    logger = logging.getLogger(name)
    logger.setLevel(level)
    
    handler = logging.StreamHandler()
    handler.setFormatter(JsonFormatter())
    logger.addHandler(handler)
    
    return logger

# Uso no job:
logger = get_logger("transform_silver")
logger.info("Processing started", extra={"extra_fields": {
    "date": "2024-01-15", "partner": "all", "job": "silver"
}})

Escrita Idempotente

Re-executar sem duplicatas — partition overwrite mode

MACKENZIE

♻️ Idempotência — Mesmo Input = Mesmo Output

Um job é idempotente quando executá-lo N vezes com os mesmos parâmetros produz o mesmo resultado que executá-lo 1 vez. Essencial para retry e backfill.
❌ NÃO IDEMPOTENTE (mode="append"):
Execução 1: Silver tem 1000 registros de 2024-01-15
Execução 2: Silver tem 2000 registros de 2024-01-15 (DUPLICOU!)
Execução 3: Silver tem 3000 registros de 2024-01-15 (3x!)

✅ IDEMPOTENTE (partition overwrite mode="dynamic"):
Execução 1: Silver tem 1000 registros de 2024-01-15
Execução 2: Silver tem 1000 registros de 2024-01-15 (SUBSTITUIU!)
Execução 3: Silver tem 1000 registros de 2024-01-15 (mesmo resultado)

# Configurar partition overwrite dinâmico
spark.conf.set("spark.sql.sources.partitionOverwriteMode", "dynamic")

# Escrita idempotente: substitui APENAS a partição processada
df_silver.write \
    .mode("overwrite") \
    .partitionBy("processing_date") \
    .parquet("/datalake/silver/vendas/")

# Resultado: só a partição de "processing_date=2024-01-15" é substituída
# Todas as outras partições permanecem intactas!
MACKENZIE

🏗️ Padrão Completo de Job Idempotente

def idempotent_write(df, output_path, partition_col, date):
    """
    Escrita idempotente: 
    1. Adiciona coluna de partição com a data de processamento
    2. Usa dynamic partition overwrite
    3. Resultado: mesma execução 1x ou Nx = mesmo output
    """
    spark = df.sparkSession
    spark.conf.set("spark.sql.sources.partitionOverwriteMode", "dynamic")
    
    df_with_partition = df.withColumn(partition_col, lit(date))
    
    df_with_partition.write \
        .mode("overwrite") \
        .partitionBy(partition_col) \
        .parquet(output_path)
    
    logger.info(f"Idempotent write complete", extra={"extra_fields": {
        "output_path": output_path,
        "partition": f"{partition_col}={date}",
        "records": df.count()
    }})

# Uso:
idempotent_write(df_silver, "/datalake/silver/vendas/", "processing_date", args.date)
# Pode rodar 10x — resultado é sempre o mesmo!
Regra de ouro: Todo job Spark deve aceitar --date e usar partition overwrite dinâmico. Retry do Airflow + backfill funcionam sem risco de duplicatas.

class TestQualityChecks:
    def test_completeness_all_filled(self, spark):
        """100% completude quando nenhum null."""
        df = spark.createDataFrame([
            ("V001", "C001", 150.0),
            ("V002", "C002", 200.0),
        ], ["order_id", "customer_id", "amount"])
        
        result = check_completeness(df, ["order_id", "customer_id"])
        assert all(r["passed"] for r in result)
    
    def test_completeness_with_nulls(self, spark):
        """Deve detectar nulls abaixo do threshold."""
        df = spark.createDataFrame([
            ("V001", None, 150.0),
            ("V002", None, 200.0),
            ("V003", "C003", 300.0),
        ], ["order_id", "customer_id", "amount"])
        
        result = check_completeness(df, ["customer_id"], threshold=0.9)
        customer_check = [r for r in result if r["column"] == "customer_id"][0]
        assert customer_check["passed"] == False
        assert customer_check["completeness"] == pytest.approx(0.333, rel=0.01)
    
    def test_uniqueness_with_duplicates(self, spark):
        """Deve detectar duplicatas."""
        df = spark.createDataFrame([
            ("V001", 150.0), ("V001", 160.0), ("V002", 200.0)
        ], ["order_id", "amount"])
        
        result = check_uniqueness(df, ["order_id"])
        assert result["passed"] == False
        assert result["duplicate_keys"] == 1
